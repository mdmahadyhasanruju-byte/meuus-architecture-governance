A. scripts/clean-next.js – 0.6kB – rmRetry .next with 5 tries + 200ms backoff – single writer ensure

Code – create apps/control/scripts/clean-next.js or control/scripts/clean-next.js:

const fs = require('fs');
const path = require('path');
const dir = path.join(__dirname, '..', '.next');
async function rmRetry(p, tries=5) {
  for (let i=0;i<tries;i++) {
    try {
      if (fs.existsSync(p)) {
        await fs.promises.rm(p, { recursive: true, force: true });
      }
      return;
    } catch(e) {
      const code = e.code || '';
      if ((code==='EPERM' || code==='EBUSY' || code==='ENOTEMPTY') && i < tries-1) {
        // Wait 200ms * (i+1) – backoff – let Defender release handle
        await new Promise(r=>setTimeout(r, 200*(i+1)));
        continue;
      }
      console.error('clean-next failed after retries:', e.message);
      throw e;
    }
  }
}
(async()=>{
  try {
    if (fs.existsSync(dir)) {
      await rmRetry(dir);
      console.log('clean-next: removed', dir);
    } else {
      console.log('clean-next: no .next to remove');
    }
  } catch(e) {
    // Fail open for build – let next build try mkdir itself – but log
    console.warn('clean-next warn:', e.message);
  }
})();
