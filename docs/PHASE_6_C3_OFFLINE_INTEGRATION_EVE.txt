async function atomicWriteWithFallback(finalPath, data, tries=3) {
  const dir = path.dirname(finalPath);
  const base = path.basename(finalPath);
  const staging = path.join(dir, `.${base}.staging.${Date.now()}.${Math.random().toString(36).slice(2)}`);
  // Ensure dir exists recursive – single writer
  await fs.promises.mkdir(dir, { recursive: true });
  // Write staging with fsync – durability
  const fd = await fs.promises.open(staging, 'w');
  try {
    await fd.writeFile(data);
    await fd.sync(); // fsync
  } finally {
    await fd.close();
  }
  // Try rename with retry – EPERM transient
  for (let i=0;i<tries;i++) {
    try {
      await fs.promises.rename(staging, finalPath);
      return; // success atomic
    } catch(e) {
      const code = e.code || '';
      if ((code==='EPERM' || code==='EBUSY' || code==='EXDEV') && i < tries-1) {
        await new Promise(r=>setTimeout(r, 100*(i+1)));
        continue;
      }
      // Fallback: copy+fsync+unlink – maintains atomicity via versioned staging + final fsync
      if (code==='EPERM' || code==='EBUSY' || code==='EXDEV') {
        try {
          // Copy
          await fs.promises.copyFile(staging, finalPath);
          // Fsync final dir – ensure copy durable – open dir fd if possible
          try {
            const dirFd = await fs.promises.open(dir, 'r');
            await dirFd.sync();
            await dirFd.close();
          } catch {}
          // Unlink staging
          await fs.promises.unlink(staging).catch(()=>{});
          return;
        } catch(e2) {
          // Clean staging on failure
          await fs.promises.unlink(staging).catch(()=>{});
          throw e2;
        }
      }
      await fs.promises.unlink(staging).catch(()=>{});
      throw e;
    }
  }
}

export async function atomicWriteWithFallback(finalPath: string, data: Buffer | string, tries=3): Promise<void> {

  const base = path.basename(finalPath);
  const staging = path.join(dir, `.${base}.staging.${Date.now()}.${Math.random().toString(36).slice(2)}`);
  await fs.mkdir(dir, { recursive: true });
  const fd = await fs.open(staging, 'w');
  try {
    await fd.writeFile(data);
    await fd.sync();
  } finally {
    await fd.close();
  }
  for (let i=0;i<tries;i++) {
    try {
      await fs.rename(staging, finalPath);
      return;
    } catch(e:any) {
      const code = e.code || '';
      if ((code==='EPERM' || code==='EBUSY' || code==='EXDEV') && i < tries-1) {
        await new Promise(r=>setTimeout(r, 100*(i+1)));
        continue;
      }
      if (code==='EPERM' || code==='EBUSY' || code==='EXDEV') {
        try {
