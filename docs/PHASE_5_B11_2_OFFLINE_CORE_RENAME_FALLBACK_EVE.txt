Current (simplified):

Proposed (fallback chain – 0.8kB):
async function atomicWriteWithFallback(tmpPath, finalPath) {
  // 1. Try rename – fast path – atomic on same volume
  for (let i=0;i<3;i++) {
    try { await fs.promises.rename(tmpPath, finalPath); return; } catch(e) {
      if (e.code==='EPERM' || e.code==='EEXIST' || e.code==='EBUSY') {
        await new Promise(r=>setTimeout(r, 100*(i+1)));
        continue;
      }
      throw e;
    }
  }
  // 2. Fallback: copy + fsync + unlink – not atomic but durable – use if rename fails after retries
  // Ensure free-space preflight already done before
  try {
    await fs.promises.copyFile(tmpPath, finalPath);
    const fd = await fs.promises.open(finalPath, 'r+');
    await fd.sync();
    await fd.close();
    await fs.promises.unlink(tmpPath);
  } catch(e) {
    // 3. Last resort: if copy fails, keep tmp and fail closed – do not leave partial
    try { await fs.promises.unlink(finalPath); } catch {}
    throw new Error('atomic-fallback-fail-closed: ' + e.message);
  }
}
