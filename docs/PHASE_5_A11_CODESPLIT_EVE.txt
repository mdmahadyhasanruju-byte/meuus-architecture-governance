import React, { useEffect, useState } from 'react';
import { getItemsAsync } from '../../lib/gameHubDatasetsAsync';
import type { DatasetType } from '../../lib/gameHubDatasets';

export function DatasetPage({ type }: { type: DatasetType }) {
  const [items, setItems] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  useEffect(() => {
    let alive = true;
    setLoading(true);
    getItemsAsync(type).then(arr => { if (alive) { setItems(arr); setLoading(false); } });
    return () => { alive = false; };
  }, [type]);
  const isDLAS = type === 'dlasFull100';
  if (loading) return <div style={{ padding: '12px', fontSize: '12px' }}>Loading {type}…</div>;
  return (
    <div style={{ padding: '12px' }}>
      <h2>{type} ({items.length})</h2>
      {isDLAS && <div style={{ fontSize: '12px', color: '#8a1b1b', margin: '6px 0' }}>Non-scoring: coverage only 0/21/50/75/100 - use DLAS Studio - key meuus_dlas_full100</div>}
      <ul style={{ margin: 0, paddingLeft: '18px' }}>{items.map((it: any) => <li key={it.id}>{it.id}{it.title ? ' - ' + it.title : ''}</li>)}</ul>
    </div>
  );
}
export default DatasetPage;

import React, { useState, Suspense, lazy, useEffect } from 'react';
import { COUNTS, getItemsAsync } from '../../lib/gameHubDatasetsAsync';
import type { DatasetType } from '../../lib/gameHubDatasets';
import { IS_DLAS_STUDIO, IS_BOOK_ZERO, IS_CAPABILITY, IS_VOCAB, IS_PILLARS } from '../../lib/flags';

const DLASStudioView = lazy(() => import('./DLASStudioView'));
const GameHubWire = lazy(() => import('./GameHubWire'));
const DataPort = lazy(() => import('./DataPort'));
const FounderShellView = lazy(() => import('./FounderShellView'));

type View = 'dlas' | 'gamehub' | 'port' | 'founder' | DatasetType;

function DatasetAsyncView({ type }: { type: DatasetType }) {
  const [items, setItems] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  useEffect(() => {
    let alive = true;
    setLoading(true);
    getItemsAsync(type).then(arr => { if (alive) { setItems(arr); setLoading(false); } });
    return () => { alive = false; };
  }, [type]);
  if (loading) return <div style={{ padding: '12px', fontSize: '12px' }}>Loading {type}...</div>;
  const isDLAS = type === 'dlasFull100';
  return (
    <div style={{ padding: '12px' }}>
      <h2>{type} ({items.length})</h2>
      {isDLAS && <div style={{ fontSize: '12px', color: '#8a1b1b', margin: '6px 0' }}>Non-scoring: coverage only 0/21/50/75/100 - key meuus_dlas_full100</div>}
      <ul style={{ margin: 0, paddingLeft: '18px' }}>{items.map((it: any) => <li key={it.id}>{it.id}{it.title ? ' - ' + it.title : ''}</li>)}</ul>
    </div>
  );
}
