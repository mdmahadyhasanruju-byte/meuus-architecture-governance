const fs = require('fs');
const path = require('path');
const dir = path.join(__dirname, '..', '.next');
async function rmRetry(p, tries=5) {
  for (let i=0;i<tries;i++) {
    try { await fs.promises.rm(p, {recursive:true, force:true}); return; } catch(e) {
      if (e.code==='EPERM' && i < tries-1) { await new Promise(r=>setTimeout(r, 200*(i+1))); continue; }
      throw e;
    }
  }
}
(async()=>{ if (fs.existsSync(dir)) await rmRetry(dir); })();
