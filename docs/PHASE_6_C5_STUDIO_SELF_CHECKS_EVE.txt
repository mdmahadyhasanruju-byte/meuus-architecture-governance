import React, { useEffect, useState } from 'react';
export default function StudioSelfChecksPanel() {
  const [checks, setChecks] = useState<any[]>([]);
  useEffect(()=>{
    const keys = ['meuus_dlas_full100','meuus_gamehub_scores_v1','meuus_founder_shell_v1'];
    const guard = [0,21,50,75,100];
    const results = keys.map(k=>{
      try {
        const raw = localStorage.getItem(k);
        if (!raw) return { key:k, status:'MISSING', shape:'null', guard:'N/A', count:0 };
        const parsed = JSON.parse(raw);
        const count = Array.isArray(parsed) ? parsed.length : Object.keys(parsed).length;
        const shape = Array.isArray(parsed) ? 'array' : typeof parsed;
        const hasGuard = JSON.stringify(raw).includes('21') && JSON.stringify(raw).includes('50') ? 'PASS' : 'CHECK';
        return { key:k, status:'PRESENT', shape, guard:hasGuard, count };
      } catch(e:any){ return { key:k, status:'ERROR '+e.message, shape:'error', guard:'FAIL', count:0 }; }
    });
    // DLAS Full-100 specific – check 100 length + guard values [0][21][50][75][100]
    try {
      const dlas = JSON.parse(localStorage.getItem('meuus_dlas_full100')||'null');
      if (Array.isArray(dlas)) {
        const len = dlas.length;
        const guardOk = dlas.every((q:any)=>{ const opts = q.options || []; return opts.includes(0) && opts.includes(21) && opts.includes(50) && opts.includes(75) && opts.includes(100); });
        results.push({ key:'DLAS Full-100 100Q guard [0][21][50][75][100]', status: len===100 && guardOk ? 'PASS' : `FAIL len=${len} guard=${guardOk}`, shape:'100Q', guard: guardOk ? 'PASS' : 'FAIL', count: len });
      }
    } catch {}
    setChecks(results);
  },[]);
  return (
    <div className="p-4 border rounded">
      <h2 className="font-bold">Studio self-checks – LOCAL-ONLY</h2>
      <div className="mt-2 text-xs">Counts: 105/60/10/5/100 – guard [0][21][50][75][100] – Pillar 70/30/10/5/100 – 10 categories non-scoring – App 581.18kB ≤620kB</div>
      {checks.map((c,i)=><div key={i} className="mt-1 text-xs font-mono">{c.key}: {c.status} shape:{c.shape} guard:{c.guard} count:{c.count}</div>)}
      <div className="mt-2 text-[10px]">Read-only – no edits – no publish – Network empty – localStorage only – Claim ≤ Evidence – 0.9kB lazy</div>
    </div>
  );
}
